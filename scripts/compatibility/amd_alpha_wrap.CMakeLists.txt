cmake_minimum_required(VERSION 3.22)
project(ModlyAlphaWrapCPU LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(CGAL REQUIRED)
add_executable(official_sample "${CGAL_DIR}/examples/Alpha_wrap_3/triangle_soup_wrap.cpp")
add_executable(alpha_check "${MODLY_PROOF_SCRIPTS}/amd_alpha_check.cpp")
add_executable(alpha_wrap "${MODLY_PROOF_SCRIPTS}/amd_alpha_wrap.cpp")
add_executable(alpha_simplify "${MODLY_PROOF_SCRIPTS}/amd_alpha_simplify.cpp")
add_executable(alpha_validate "${MODLY_PROOF_SCRIPTS}/amd_alpha_validate.cpp")
foreach(t official_sample alpha_check alpha_wrap alpha_simplify alpha_validate)
 target_link_libraries(${t} PRIVATE CGAL::CGAL)
 target_compile_definitions(${t} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(${t} PRIVATE /bigobj /O1)
endforeach()
